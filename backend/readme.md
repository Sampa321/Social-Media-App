# cookie - parser 
  - a middleware that parse cookies from the request and make accessible via req.cookies
  - app.use(cookieParser())

# what is  cookie
   - small piece of data stored in the browser by server
   - it is automatically sent with every request to server

# what is token
   - a unique string generated by server after login that proves user identity 
   token -> client store -> server verify 

# steps 
- login request by client 
- server check is login credentials true or not 
- server create jwt token
- add that to cookie
- send back response 


# for every protected request 
- server will decrypt the cookie(token)
- try to find the real user behind request 
- serve according to the user 


# logout means 
  - logout means deleting the cookie from user's browser 

# how to send cookie
- res.cookie('token',"<actual token>")

# how to access 
- req.cookies (but use cookie parser middleware first)


# jwt token
 - jwt means json web token
 - jwt is a compact , self contained token used for authentication

# for creating token 
 - jwt.sign(payload , secret)
# for getting the payload from token
 - jwt.verify(token , secret)



# why auth middleware ??
- because we want all route protected except login and signup


# how to expire jwt ??
 - jwt.sign(data,secret , {expiresIn : '7d'})


# how to expire cookie ?
res.cookie("token",<token>,{expires : new Date( Date.now()+ miliseconds )})


# why expire token ??
- stolen token protection
- force re-authentication

// For create user and signup
app.post("/signup", async function(req, res) {
    try{
        const {name, email, password} = req.body;
        const user = new User({name : name, email : email, password : password});
        const createUser = await user.save();
        res.json({
            message : "User Created successfully",
            user : createUser
        });
    }catch(error)
    {
        res.json({
            error : error.message
        })
    }
    
});
